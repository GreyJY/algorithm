进制转换思想
10进制转换其他进制
static String toBase(long number,int base){
if(number==0){
return"0";
}
String digits="0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
StringBuilder result = new StringBuilder();
while(number>0){
int remainder=(int)(number%base);
result.append(digits.charAt(remainder));
number/=base;
}
return result.reverse().toString();
}

其他进制转换为十进制
static long fromBase(String value,int base){
long result =0;
for(int i=0;i<value.length();i++){
char c = value.charAt(i);
int digit;
if(c>='0'&&c<='9'){
digit=c-'0';
}else{digit =c-'A'+10;}
if(digit>=base){
throw new ILLegalArgumentException("数字不属于该进制")；
}
result = result*base+digit;
}

return result;
}

static boolean isLeapYear(int year){
return year%400==0||(year%4==0&&year%100!=0);
}

日期第二天
static int[] nextDay(int year, int month, int day) {
    day++;
    if (day > daysOfMonth(year, month)) {
        day = 1;
        month++;
        if (month > 12) {
            month = 1;
            year++;
        }
    }
    return new int[]{year, month, day};
}

判断某日期是当年的第几天
static int dayOfYear(int year,int month,int day){
int result=day;
for(int m=1;m<month;m++){
result+=dayOfMonth(year,m);
}
return result;
}
